# Price Tiers Sort
# Sorts cards by USD value into price brackets.
# High-value cards first so they get their own bins.
# Cheap cards get multiple overflow bins since there are more of them.
#
# Optional: set "limit: 300" to cap each bin at 300 cards
# (matches config.DEFAULT_BIN_CAPACITY — roughly when the Z-probe
# fullness check trips on the physical bins).
# When a bin fills up, cards overflow to the next matching bin.

bins: 8
fallback: 8

bin1: usd>=10
bin2: usd>=5 usd<10
bin3: usd>=1 usd<5
bin4: usd<1
bin5: usd<1
bin6: usd<1
bin7: usd<1
# bin8 = fallback (cards with no price data or all bins full)
